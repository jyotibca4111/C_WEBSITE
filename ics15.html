Enter a number from user and print first digit of number ?

#include<stdio.h>
void main()
{
int num,firstdigit;
printf("Enter the number:");
scanf("%d",&num);
if(num<0)
num=-num;
while(num>=10)
{
num=num/10;
}
firstdigit=num;
printf("the first digit is:%d\n",firstdigit);
}

Output:-

Enter a number: 903
first digit: 9
