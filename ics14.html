Enter two numbers from users, one is base and second is power. Write the result ?

#include<stdio.h>
void main()
{
double base, power,result;
printf("Enter base number:");
scanf("%lf",&base);
printf("Enter power:");
scanf("%lf",&power);
result=(base,power);
printf("%2lf raised to the power%2lf=%2lf\n",base,power,result);
}
