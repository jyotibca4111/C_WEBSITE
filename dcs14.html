Enter a number from the user if the user enters 30 then print the month in which we have 30 days. If the enter 31 then print the month 31 days for 28, 29 print the month else everything is invalid ?

#include <stdio.h>
int main() {
int days;
printf("Enter number of days: ");
scanf("%d", &days);
switch(days){
case 31:
printf("Months: January, March, May, July, August, October, December"); break;
case 30:
printf("Months: April, June, September, November"); break;
case 28:
printf("Month: February"); break;
case 29:
printf("Month: February"); break;
default:
printf("Invalid number of days");}
}

Output:-

Enter number of days:28
month:February
