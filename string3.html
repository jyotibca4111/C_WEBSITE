Enter a string from user and convert into sentence case ?

#include<stdio.h>
void main()
{
char str[50];
int i=0;
printf("enter a string: ");
fgets(str,sizeof(str),stdin);
while(str[i]!='\0')
{
str[i]=tolower(str[i]);
i++;
}
if(str[0]>='a'&&str[0]<='z')
{
str[0]=toupper(str[0]);
}
printf("String in sentence case : %s\n",str);
}

Output:-

enter a string: hELLO
string in sentence case: Hello
