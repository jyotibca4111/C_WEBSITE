Enter two numbers from user and print range from first number to second number ?

#include<stdio.h>
void main()
{
int num1,num2,i;
printf("Enter first number:");
scanf("%d",&num1);
printf("Enter second number:");
scanf("%d",&num2);
printf("Numbers from%d to %d are\n",num1,num2);
for(i=num1;i<=num2;i++)
{
printf("%d",i);
}
printf("\n");
}

Output:-

Enter first number:5
Enter second number: 10
number from 5 to 10 are:
5 6 7 8 9 10
