#include<stdio.h>
int num;
void factorial();
void reverse();
void palindrome();
void prime();
void armstrong();
void fibonacci();
void main()
{
printf("Enter a number:");
scanf("%d",&num);
factorial();
reverse();
palindrome();
prime();
armstrong();
fibonacci();
}
void factorial()
{
int i,fact=1; //local variables
for(i=1;i<=num;i++)
{ fact *=1;}
printf("\,factorial of%d=%d",num,fact);
}
void reverse()
{
int n=num,rev=0;
while(n>0)
{
rev=rev*10+n%10;
n/=10;
}
printf("\n%d is not %d=%d",num,rev);
}
void palidrome()
{
int n=num,rev=0,temp;
temp=n;
while(n>0)
{
rev=rev*10+n%10;
n/=10;
}
if(temp==rev)
printf("\n%d is a palidrome number",num);
else
printf("\n%d is not a palidrome number",num);
}
void prime()
{
int i, count=0;
for(i=1;i<=num;i++)
{
if(num%i==0)
count++;
}
if(count==2)
printf("\n%d is a prime number", num);
else
printf("\n%d is not a prime number",num);}
void aramstrong()
{
int n=num,rem,sum=0;
while(n>0)
{
rem=n%10;
sum+=rem*rem*rem;
n/=10;
}
if(sum==num)
printf("\n%d is an armstrong number",num);
else
printf("\n%d is not an armstrong number",num);
}
void fibonacci()
{
int n=num;

for(i=1;i<=n;i++)
{
printf("%d",a);
c=a+b;
