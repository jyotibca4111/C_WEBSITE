Enter a string from user and convert into toggle case ?

#include<stdio.h>
void main()
{
char str[50];
int i=0;
printf("enter a string: ");
fgets(str,sizeof(str),stdin);
while(str[i]!='\0')
{ if(str[i]>='a'&&str[i]<='z')
{
str[i]=toupper(str[i]);
}
else if(str[i]>='A'&&str[i]<='Z')
{
str[i]= tolower(str[i]);
}
i++;
}
printf("String in toggle case: %s\n",str);
}

Output:-

enter a string: hELLO wORLD
string in toggle case:Hello World
