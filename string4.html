Enter a string from user and convert title case ?

#include<stdio.>
void main()
{
char str[50];
int i=0;
printf("enter a string: ");
fgets(str,sizeof(str),stdin);
while(str[i]!='\0')
{
str[i]=tolower(str[i]);
i++;
}
if(str[0]>='a'&&str[0]<='z')
{
str[0]=toupper(str[0]);
}
while(str[i]!='\0')
{
if(str[i]==' '&&str[i+1]>='a'&&str[i+1]<='z')
{
str[i+1]=toupper(str[i+1]);
}
i++;
}
printf("String in title case: %s\n",str);
}

Output:-

enter a string: hello world
string in title case: Hello World
