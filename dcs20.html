Enter the year from user and check its leap year or not ?

#include<stdio.h>
void main()
{
int year;
printf("enter the year:");
scanf("%d",&year);
if((year%400==0)||(year%4==0&&year%100!=0))
{
printf("%d is IT IS A LEAP YEAR",year);
}
else
printf("%d is NOT A LEAP YEAR",year);
}
