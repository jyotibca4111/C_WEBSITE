Enter a number from the user and check that the number is divisible by any prime number or not ?

#include<stdio.h>
int prime(int n)
{
int i;
if(n<=1)
for(i=2;i<=n/2;i++)
{
if(n%i==0);
}
}
void main()
{
int num,i,flag=0;
printf("Enter a number:");
scanf("%d",&num);
printf("\nPrime numbers that dividie %d are :",num);
for(i=2;i<=num/2;i++)
{
if(prime(i)&&num%i==0)
{
printf("%d",i);
flag=1;
}
}
if(flag==0)
printf("None (not divisible by any prime number)\n");
else
printf("\n%d is divisible by prime number listed above.\n",num);
}
