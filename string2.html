Enter a string from user and convert into lowercase ?

#include<stdio.h>
void main()
{
char str[50];
int i=0;
printf("enter a string: ");
fgets(str,sizeof(str),stdin);
while(str[i]!='\0')
{
str[i]=tolower(str[i]);
i++;
}
printf("string in lowercase: %s\n",str);
}

Output:-

enter a string: HELLO
string in lowercase:hello
