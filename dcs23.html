Write a program to calculate area and perimeter of rectangle ?

#include<stdio.h>
void main()
{
int l,b,a,perimeter;
printf("Enter the length of the rectangle:");
scanf("%d",&l);
printf("Enter the breadth of the rectangle:");
scanf("%d",&b);
a=l*b;
perimeter=2*l+b;
printf("The area of rectangle is:%d\n",a);
printf("The perimeter of rectangle is:%d",perimeter);
}
