Enter a number from the user and check it is a prime number or not ?

#include<stdio.h>
void main()
{
int num,i,flag=0;
printf("enter a number:");
scanf("%d",&num);
if(num<=1)
{
printf("%d is not a prime number.\n",num);
return 0;
}
for(i=2;i<=num/2;i++)
{
if(num%1==0)
{
flag=1;
break;
}
}
if(flag==0)
printf("%d is a prime number.\n",num);
else
printf("%d is not a prime number.\n",num);
}

output:-

enter a number : 7
7 is a prime number
