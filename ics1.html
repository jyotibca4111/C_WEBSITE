void binary(int n)
{
int binary[32];
int i=0;
if(n==0)
{
printf("0");
}
while(n>0)
{
binary[i]=n%2;
n=n/2;
i++;
}
for(int j=i-1;j>=0;j--)
{
printf("%d",binary[j]);
}
printf("\n");
}
void main()
{
int num;

printf("Enter the number:");
scanf("%d",&num);
binary(num);
printf("Binary of%d is:",num);
}
