Enter a character from user and it is in upper case, lower case, special characters or numbers ?

#include<stdio.h>
void main()
{
char ch;
printf("enter any character");
scanf("%c",&ch);
if(ch>='A'&&ch<='Z')
printf("IT IS AN UPPERCASE CHARACTER");
else if(ch>='a'&&ch<='z')
printf("IT IS LOWERCASE CHARACTER");
else if(ch>=0&&ch<=9)
printf("IT IS NUMBERS");
else
printf("IT IS A SPECIAL CHARACTER");
}

Enter any charcter:S
