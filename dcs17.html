Enter three numbers from user and which is greater ?

#include <stdio.h>
int main() {
int a, b, c;
printf("Enter three numbers: ");
scanf("%d %d %d", &a, &b, &c);
if(a>b && a>c) printf("%d is greatest", a);
else if(b>c) printf("%d is greatest", b);
else printf("%d is greatest", c);
}

Output:-

Enter three number: 15,20,25
25 is greatest
