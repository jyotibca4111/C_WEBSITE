Calculate simple interest = (P x R x T)/ 100

#include<stdio.h>
void main()
{
float p,r,t,intrest;
printf("Enter the principal rate:");
scanf("%f",&p);
printf("Enter the rate of intrest:");
scanf("%f",&r);
printf("Enter the time:");
scanf("%f",&t);
intrest=p*r*t/100;
printf("The simple intrest is: %f\n",intrest);
}

Output:-

enter principal: 1000
enter rate of interest: 5
enter the time: 2
