Enter a string from user and convert in uppercase ?

#include<stdio.h>
void main()
{
char str[50];
int i=0;
printf("enter a string: ");
fgets(str,sizeof(str),stdin);
while(str[i]!='\0')
{
str[i]=toupper(str[i]);
i++;
}
printf("string in uppercase: %s\n",str);
}

Output:-

Enter a string: c programming
string in uppercase : C PROGRAMMING
